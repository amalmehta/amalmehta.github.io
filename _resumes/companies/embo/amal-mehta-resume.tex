\section*{Experience}

\role{Independent RL \& Robotics Research}{Sept 2025 -- Present}{Self-directed projects, each with public code and a write-up at \href{https://amalmehta.com/}{amalmehta.com}}{San Francisco Bay Area}
\begin{itemize}
\item \href{https://amalmehta.com/RewardModeling/}{\textbf{Reward modeling:}} learned rewards from preferences and demonstrations across five reward shapes; EPIC distance predicted policy regret better than raw reward correlation (0.61 vs.\ 0.54).
\item \href{https://amalmehta.com/long_term_horizon_planning/}{\textbf{Reward hacking (toy code repair):}} RL on visible tests raised the pass rate to 58\% while true correctness fell to 2\%; choosing which episodes to audit held false passes to 4\%, against 24\% for random audits.
\item \href{https://amalmehta.com/DroneAutonomy/}{\textbf{Quadrotor adaptation:}} a simulator with randomized payload, wind and motor wear; a learned residual cut tracking error to 7.3\,cm, against 14.0\,cm for tuned adaptive control.
\end{itemize}

\role{Deep Learning Scientist}{Sept 2021 -- Aug 2025}{PhysicsAI (DARPA-funded)}{Pacifica, CA (remote)}
\begin{itemize}
\item Turned synthetic data generation for drone perception into a closed RL loop: agents (TD3, PPO, DDPG) chose an image generator's settings, a detector trained on the images, and its accuracy on real drone footage was the reward.
\item Built the PEARL meta-RL codebase for a fighter-jet simulation: the agent infers a latent embedding of each task from off-policy experience, so it adapts to a new task in a few tries. Ran its multi-week GPU training.
\item Implemented domain adaptation from scratch to keep the detector accurate with few real labels and mostly synthetic data (10--15\,GB).
\item Built and benchmarked the GAN pipelines (DCGAN, InfoGAN, FineGAN) that generated the detector's training images.
\end{itemize}

\role{Deep RL Researcher}{Aug 2023 -- Mar 2024}{Hybrid Robotics Lab, UC Berkeley --- adv.\ Prof.\ Koushil Sreenath}{Berkeley, CA}
\begin{itemize}
\item Trained one PPO policy in IsaacGym that took a real Unitree Go1 from tracking a soccer ball to kicking it at a target; about 60\% accuracy on hardware.
\item Designed its three-stage reward; prototyped RLPD to fine-tune on real robot data, mixing offline data with online replay.
\end{itemize}

\role{Research Assistant}{June 2020 -- Dec 2022}{Video \& Image Processing Lab, BAIR --- adv.\ Prof.\ Avideh Zakhor}{UC Berkeley}
\begin{itemize}
\item Built a deep learning pipeline that outlines melanoma in gigapixel tissue scans (F1 0.44 vs.\ 0.14 prior best); published at SPIE Medical Imaging 2023.
\end{itemize}
